# Налаштування Gemini Dictate. Після зміни - перезапустити програму.

# --- Рушій ---
# local = Parakeet TDT 0.6B v3 на цьому ПК, без інтернету (основний з 13.09.2026)
# stt   = хмарна модель розпізнавання через OpenRouter (резерв)
# chat  = Gemini через chat completions (старий запасний варіант)
ENGINE=local

# --- Локальна модель ---
# Заміряно 13.09.2026 на 40 архівних фразах (i9-12900F, лише CPU):
#   локально  медіана 0.16 с, p95 0.34 с, найгірша 0.48 с, RAM +730 МБ
#   хмара     медіана ~1.0 с,  p95 5.3 с,  хвости до 22 с
# Модель лежить у папці models\ (~700 МБ), вантажиться ~2 с при старті.
LOCAL_MODEL=nemo-parakeet-tdt-0.6b-v3
LOCAL_QUANT=int8
# 8 потоків - оптимум. 14 і більше ПОВІЛЬНІШЕ: робота розповзається на
# E-ядра, і модель чекає найповільніше. 2-4 потоки теж швидші за хмару.
LOCAL_THREADS=8
# 1 = якщо локальна модель ще вантажиться або впала - фраза йде у хмару.
# Примусова мова (Ctrl+Alt+L) завжди йде у хмару: локальна модель
# мову не приймає. Це і є шлях для німецької.
LOCAL_FALLBACK=1

# --- Числа цифрами ---
# Локальна модель пише "сім тисяч п'ятсот гривень". З NUMBERS=1 це стає
# "7500 гривень". Правила (numwords.py): переводиться, якщо число >= 10,
# або з двох і більше слів, або далі йде одиниця (кВт, %, грн, метри,
# години, штуки). "Один момент", "через два дні", дати ("двадцять шостого
# року") і діапазони ("місяць-півтора") лишаються словами.
# Кожне перетворення пишеться в dictate.log рядком "numbers:".
NUMBERS=1

# Модель для ENGINE=stt. Заміряно 09.09.2026 на твоїх власних фразах
# (медіана з 5 запитів, вартість за 5 фраз ~35 сек аудіо):
#   openai/whisper-large-v3-turbo  1.34s  $0.00039  цифри "20 кВт" вірно,
#                                                   бренди хибно (Pylon Edge)
#   openai/gpt-4o-transcribe       1.21s  $0.00220  бренди ідеально
#                                                   (Victron Energy, Pylontech),
#                                                   але числа словами
#   google/chirp-3                 2.82s  $0.01040  і бренди, і цифри вірно,
#                                                   але найповільніша
#   microsoft/mai-transcribe-2     1.16s  $0.00108  посередині
#   deepgram/nova-3                1.44s  $0.00252  бренди хибно
#   openai/gpt-4o-mini-transcribe  1.00s  $0.00110  зривається в білоруську
#   openai/whisper-large-v3        4.40s  $0.00026  повільна
#   qwen3-asr-flash                3.17s  $0.00119  бренди кирилицею
#   mistralai/voxtral-mini         не працює (помилка 400)
STT_MODEL=openai/whisper-large-v3-turbo

# --- Мова ---
# ПОРОЖНЄ значення = модель сама визначає мову кожної фрази.
# Саме так і треба, якщо диктуєш і українською, і російською.
# Якщо вписати код (uk / ru / en / de) - модель буде ПРИМУСОВО писати
# цією мовою, навіть коли ти говориш іншою.
LANGUAGE=

# Мови, якими ти реально диктуєш. Якщо модель визначила щось інше
# (білоруську, польську, болгарську - вони близькі й плутаються),
# фраза автоматично переробляється як FALLBACK_LANG.
ALLOWED_LANGS=uk,ru,en,de
FALLBACK_LANG=uk

# Німецьку всередині українських/російських фраз автовизначення плутає.
# Ця клавіша перемикає примусову мову по колу: авто -> uk -> ru -> de -> en.
# Поточна мова показується в плашці.
LANG_HOTKEY=ctrl+alt+l
LANG_CYCLE=,uk,ru,de,en

# --- Швидкість мережі ---
# Порядок провайдерів OpenRouter. Заміряно 09.09.2026 на твоїх записах:
#   DeepInfra: медіана 2.26с, максимум 3.25с - стабільний
#   Groq:      медіана 4.70с, максимум 81.8с - непередбачуваний
# Дубль-запит завжди йде до ІНШОГО провайдера зі списку.
PROVIDER_ORDER=DeepInfra,Groq
# Якщо відповіді немає стільки секунд - надсилається другий такий самий
# запит паралельно, береться той, що відповів першим. 0 = вимкнути.
HEDGE_AFTER=1.5

# --- Нарізка на фрази ---
# Пауза (сек), після якої фраза вважається завершеною.
SILENCE_TAIL=0.45
# Пауза на подумати - не кінець речення. Якщо мова поновилася раніше
# ніж через стільки секунд, крапка не ставиться, а наступне слово йде
# з малої літери. Більше = охочіше склеює. 0 = вимкнути склейку.
JOIN_WINDOW=1.1

# --- Автовимкнення ---
# Через скільки секунд тиші диктування вимкнеться саме. Захист від того,
# щоб увімкнений мікрофон зловив чужу розмову і вставив її у чат.
# 0 = ніколи не вимикати.
IDLE_OFF=120
# За скільки секунд до вимкнення плашка починає гаснути й попереджати.
IDLE_WARN=10
# Мінімальна довжина фрази (сек).
MIN_PHRASE=0.7
# Скільки фраз розпізнавати одночасно. Порядок тексту завжди зберігається,
# паралеляться лише очікування мережі. 1 = як було раніше.
MAX_PARALLEL=3
# Затримки навколо вставки тексту. Якщо в якійсь програмі текст почне
# вставлятися неповністю - збільш до 0.10-0.15.
PASTE_DELAY=0.05

# --- Захист від галюцинацій на шумі ---
# 13.09.2026: фіксований поріг 0.030 з'їв 38 шматків справжньої мови,
# коли рівень мікрофона впав утричі (піки 0.013-0.030 при шумі 0.0002).
# Тепер поріг = max(MIN_PEAK, шум_кімнати * PEAK_SNR). У тихій кімнаті
# пропускає тиху мову, у гучній - суворіший за старі 0.030.
MIN_PEAK=0.010
PEAK_SNR=8.0
# Скільки секунд у фразі має бути реального звуку.
MIN_VOICED=0.32
# Поріг тиші для окремого блоку (RMS). 13.09: тиха мова з піком 0.014
# має RMS ~0.003 і при 0.004 взагалі не вважалася звуком ("only 0.03s
# voiced"). Шум кімнати 0.0002, тож 0.0015 - все ще у 7 разів вище шуму.
ABS_FLOOR=0.0015
SNR_FACTOR=3.5
# Максимальне автопідсилення. Було 12 - при тихому мікрофоні впиралося
# в стелю у 72% фраз, і в модель ішов занадто тихий звук.
MAX_GAIN=25.0

# --- Гарячі клавіші ---
# Можна перелічити кілька через кому - працюватимуть усі одночасно.
# Безпечні одиночні клавіші (нічого не ламають):
#   scroll lock, pause, insert, f13..f24
# Клавіші, які вже щось роблять у Windows (потребують SUPPRESS):
#   f1 (довідка), f2 (перейменування, Excel), f12 (інструменти браузера)
HOTKEY=num lock, scroll lock, pause

# Подвійне натискання цієї клавіші теж вмикає/вимикає диктування.
# ctrl (за замовчуванням), shift, alt. Порожнє значення = вимкнути.
# Ctrl+C, Ctrl+V та інші комбінації НЕ спрацьовують: рахується лише
# натискання Ctrl окремо, без інших клавіш.
DOUBLE_TAP=ctrl
# Максимальний проміжок між двома натисканнями (сек).
TAP_GAP=0.45
# Якщо Ctrl тримали довше - це не "тап", а утримання.
TAP_MAX_HOLD=0.40

# Клавіші, які треба перехопити повністю, щоб програма під фокусом їх не бачила.
# Приклад: SUPPRESS=f2  - тоді F2 вмикає диктування, але перестає
# перейменовувати файли в Провіднику і редагувати клітинку в Excel.
# Scroll Lock перехоплюється за замовчуванням: інакше Windows вмикав би
# сам режим Scroll Lock, а він змінює поведінку стрілок в Excel.
# УВАГА: suppress ламає обробку всіх клавіш у бібліотеці keyboard.
# Замість нього Num Lock / Scroll Lock повертаються у попередній стан
# автоматично одразу після натискання. Залишай це поле порожнім.
SUPPRESS=

CHAT_MODEL=google/gemini-2.5-flash
